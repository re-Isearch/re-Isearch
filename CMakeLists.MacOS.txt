# ─────────────────────────────────────────────
# Options
# ─────────────────────────────────────────────

# ─────────────────────────────────────────────
# Paths (override on command line with -D)
# ─────────────────────────────────────────────
set(BERKELEY_DB_ROOT "/opt/homebrew/Cellar/berkeley-db/18.1.40_2"
    CACHE PATH "Berkeley DB root")
set(LIBMAGIC_ROOT "/opt/homebrew/Cellar/libmagic/5.45"
    CACHE PATH "libmagic root")

    CACHE PATH "OpenSSL root (for autoremote plugin)")
set(SCHMATE_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/../Schmate/include"
    CACHE PATH "Schmate/HNSW include dir")

set(INSTALL_BIN_DIR   "/usr/local/bin"       CACHE PATH "Executable install dir")
set(INSTALL_LIB_DIR   "/usr/local/lib/ib"    CACHE PATH "Library install dir")

# ─────────────────────────────────────────────
# Compiler flags
# ─────────────────────────────────────────────
add_compile_options(
    -m64
    -O3
    -fvectorize
    -ffast-math
    -ftree-vectorize
    -fpic
)

    BSD
    WWW_SUPPORT
    __CSTRING__

    DO_BUILD_IB64=1
)

if(USE_JEMALLOC)
    find_library(JEMALLOC_LIB jemalloc HINTS /opt/homebrew/lib REQUIRED)

# ─────────────────────────────────────────────
# Executables
# ─────────────────────────────────────────────

    add_executable(${exe} ${SRC_DIR}/${exe}.c)
    target_link_libraries(${exe} PRIVATE ${COMMON_LINK_LIBS})
    set_target_properties(${exe} PROPERTIES
        COMPILE_FLAGS "-m64"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/../bin"
    )

# ─────────────────────────────────────────────
# autoremote plugin (needs OpenSSL)
# ─────────────────────────────────────────────
find_library(SSL_LIB   ssl   HINTS "${OPENSSL_ROOT}/lib")
find_library(CRYPTO_LIB crypto HINTS "${OPENSSL_ROOT}/lib")

if(SSL_LIB AND CRYPTO_LIB)
    add_library(autoremote MODULE ${DOCTYPE_DIR}/autoremote.cxx)
    target_include_directories(autoremote PRIVATE
        ${SRC_DIR} ${DOCTYPE_DIR}
        "${OPENSSL_ROOT}/include"
    )
    target_link_libraries(autoremote PRIVATE
        ${COMMON_LINK_LIBS} ${SSL_LIB} ${CRYPTO_LIB}
    )
    set_target_properties(autoremote PROPERTIES
        PREFIX ""
        SUFFIX ".dylib"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/../lib/plugins"
    )
endif()
